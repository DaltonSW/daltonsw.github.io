= true -}} {{/* same last-activity rule as game-last-played.html, so rows stay in the same order */}} {{- $activityEnd := $entry.started -}} {{- $entryOpen := false -}} {{- if gt (len $entry.sessions) $entry.sessions) 0 -}} {{- $subgroupStack = merge $subgroupStack (dict $sub false) -}} {{/* connector bridges are synthesized client-side by games-timeline.ts */}} {{- range $sIdx, $session := (sort $items = $items | append (dict "id" (printf "%s--session-%d" $baseID $sIdx) "group" $slug "subgroup" $sub "type" "range" "className" (printf "pt-session pt--%s" $status) "content" "" "title" $sTitle -}} {{- end -}} {{- if $hasItems -}} {{/* games with a genuinely open session float to the top ("1" outranks "0") */}} {{- $tier := "0" -}} {{- if $groupHasOpen -}}{{- $tier = "1" -}}{{- end -}} {{- $groupCandidates = $groupCandidates | append (dict "id" $slug "content" (printf "%s" $game.RelPermalink $game.Title) "subgroupStack" $subgroupStack "sortKey" (printf "%s-%s" $tier $groupLast) ) -}} {{- end -}} {{- end -}} {{- $groups := slice -}} {{- range sort $groupCandidates "sortKey" "desc" -}} {{- $groups = $groups | append (dict "id" .id "content" .content "subgroupStack" .subgroupStack) -}} {{- end -}}
