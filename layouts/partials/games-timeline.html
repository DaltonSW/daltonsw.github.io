{{/* Aggregate gameplay history timeline. Builds vis-timeline `groups` (one per game) and `items` (one per playthrough entry — a solid range item, or for sessions-bearing entries a low-opacity connector plus one item per session sharing a subgroup so they overlap instead of stacking) from every game's merged entries (see playthrough-entries.html), then renders the container, JSON islands, and the js.Build + vendored-CSS pipeline. Included from layouts/games/taxonomy.html; context is the games taxonomy page. */}} {{- $fmt := "2006-01-02" -}} {{- $groupCandidates := slice -}} {{- $items := slice -}} {{- range .Pages -}} {{- $game := . -}} {{- $slug := $game.File.ContentBaseName -}} {{- $entries := partial "playthrough-entries.html" $game -}} {{- if eq (len $entries) 0 -}} {{- with $game.Params.started -}} {{- $entries = slice (dict "written" false "started" . "finished" $game.Params.finished "status" $game.Params.status "rating" $game.Params.rating "notes" nil "sessions" slice ) -}} {{- end -}} {{- end -}} {{- $subgroupStack := dict -}} {{- $hasItems := false -}} {{- $groupStart := "" -}} {{- range $idx, $entry := $entries -}} {{- if $entry.started -}} {{- $hasItems = true -}} {{- if or (eq $groupStart "") (lt $entry.started $groupStart) -}} {{- $groupStart = $entry.started -}} {{- end -}} {{- $sub := printf "pt-%d" $idx -}} {{- $baseID := printf "%s--%s" $slug $sub -}} {{- $status := $entry.status | default "backlog" -}} {{- $start := $entry.started | time.Format $fmt -}} {{- $end := "" -}} {{- with $entry.finished -}}{{- $end = (. | time.Format $fmt) -}}{{- end -}} {{- $titleParts := slice ($entry.started | time.Format "Jan 2, 2006") -}} {{- if $end -}} {{- $titleParts = $titleParts | append ($entry.finished | time.Format "Jan 2, 2006") -}} {{- else -}} {{- $titleParts = $titleParts | append "ongoing" -}} {{- end -}} {{- with $entry.rating -}}{{- $titleParts = $titleParts | append (printf "%v/10" .) -}}{{- end -}} {{- $title := printf "%s · %s" ($status | title) (delimit $titleParts " – ") -}} {{- with $entry.notes -}} {{- $title = printf "%s
%s" $title (. | markdownify | plainify | truncate 160) -}} {{- end -}} {{- $permalink := "" -}} {{- if $entry.written -}}{{- $permalink = $entry.permalink -}}{{- end -}} {{- if gt (len $entry.sessions) 0 -}} {{- $subgroupStack = merge $subgroupStack (dict $sub false) -}} {{- $items = $items | append (dict "id" (printf "%s--connector" $baseID) "group" $slug "subgroup" $sub "type" "range" "className" (printf "pt-connector pt--%s" $status) "content" "" "start" $start "end" $end ) -}} {{- range $sIdx, $session := (sort $entry.sessions "started" "asc") -}} {{- $sStart := $session.started | time.Format $fmt -}} {{- $sEnd := "" -}} {{- with $session.finished -}}{{- $sEnd = (. | time.Format $fmt) -}}{{- end -}} {{- $items = $items | append (dict "id" (printf "%s--session-%d" $baseID $sIdx) "group" $slug "subgroup" $sub "type" "range" "className" (printf "pt-session pt--%s" $status) "content" "" "title" $title "start" $sStart "end" $sEnd ) -}} {{- end -}} {{- else -}} {{- $items = $items | append (dict "id" $baseID "group" $slug "subgroup" $sub "type" "range" "className" (printf "pt pt--%s" $status) "content" "" "title" $title "permalink" $permalink "start" $start "end" $end ) -}} {{- end -}} {{- end -}} {{- end -}} {{- if $hasItems -}} {{- $groupCandidates = $groupCandidates | append (dict "id" $slug "content" (printf "%s" $game.RelPermalink $game.Title) "subgroupStack" $subgroupStack "sortKey" $groupStart ) -}} {{- end -}} {{- end -}} {{/* Row order matches the list below: most-recently-started game at the top. */}} {{- $groups := slice -}} {{- range sort $groupCandidates "sortKey" "desc" -}} {{- $groups = $groups | append (dict "id" .id "content" .content "subgroupStack" .subgroupStack) -}} {{- end -}}
