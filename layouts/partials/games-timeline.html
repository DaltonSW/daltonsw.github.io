{{/* Aggregate gameplay history timeline. Builds vis-timeline `groups` (one per game) and `items` (one per playthrough entry — a solid range item, or for sessions-bearing entries a low-opacity connector plus one item per session sharing a subgroup so they overlap instead of stacking) from every game's merged entries (see playthrough-entries.html), then renders the container, JSON islands, and the js.Build + vendored-CSS pipeline. Included from layouts/games/taxonomy.html; context is the games taxonomy page. */}} {{- $fmt := "2006-01-02" -}} {{/* Statuses that legitimately have no finish line yet and should keep stretching to "today" — everything else (paused, dropped, backlog, finished, mastered) gets capped at its real last-known activity instead, so a paused game's bar stops implying it's still being played. */}} {{- $openEndedStatuses := slice "playing" "ongoing" "multiplayer" "software" -}} {{- $groupCandidates := slice -}} {{- $items := slice -}} {{- $earliestStart := "" -}} {{- range .Pages -}} {{- $game := . -}} {{- $slug := $game.File.ContentBaseName -}} {{- $entries := partial "playthrough-entries.html" $game -}} {{- if eq (len $entries) 0 -}} {{/* hugo.toml's [frontmatter] config wires started/finished as .Date/.Lastmod sources, so Hugo hands back a time.Time instead of a string for whichever page actually uses one that way — normalize through time.Format so later string comparisons never compare across mismatched types. */}} {{- with $game.Params.started -}} {{- $finished := "" -}} {{- with $game.Params.finished -}}{{- $finished = (. | time.Format "2006-01-02") -}}{{- end -}} {{- $entries = slice (dict "written" false "started" (. | time.Format "2006-01-02") "finished" $finished "status" $game.Params.status "rating" $game.Params.rating "notes" nil "sessions" slice ) -}} {{- end -}} {{- end -}} {{- $subgroupStack := dict -}} {{- $hasItems := false -}} {{- $groupLast := "" -}} {{- range $idx, $entry := $entries -}} {{- if $entry.started -}} {{- $sub := printf "pt-%d" $idx -}} {{- $baseID := printf "%s--%s" $slug $sub -}} {{/* Fall back to the game's own overall status, not a bare "backlog" — an entry with real dates but a blank status (a data gap, e.g. a split-off entry that never got one filled in) is still a real playthrough and shouldn't read as "never played." Mirrors game-year-rows.html's fallback for the same blank-status case. */}} {{- $gameStatus := $game.Params.status -}} {{- $status := $entry.status | default $gameStatus -}} {{/* ongoing, multiplayer, and software exist only in the game-level vocabulary — those entries keep a single playthrough whose status stays "playing" while its sessions accumulate (see tools/gamelog KNOWN-ISSUES). Colouring the bar from the entry would render an endless sandbox — or a paint program — identically to an active playthrough, so the game's own status wins for these three. */}} {{- if in (slice "ongoing" "multiplayer" "software") $gameStatus -}} {{- $status = $gameStatus -}} {{- end -}} {{/* A dated entry can still resolve to "backlog" by hand-editing status: backlog directly onto a playthroughs.yaml entry (nothing in the CLI ever produces this — see tools/gamelog/KNOWN-ISSUES). "This is on the backlog from 2 years ago" doesn't make sense on a timeline of when things were actually played, so it's excluded entirely rather than drawn with backlog's dim colour. */}} {{- if ne $status "backlog" -}} {{- $hasItems = true -}} {{/* Last known activity: an entry's finished date once it has one, otherwise the started date of whatever's still open — its last session if it has any, else the entry itself. Mirrors game-last-played.html so chart rows and the list below stay in the same order. */}} {{- $activityEnd := $entry.started -}} {{- if gt (len $entry.sessions) 0 -}} {{- $lastSession := index (sort $entry.sessions "started" "asc") (sub (len $entry.sessions) 1) -}} {{- $activityEnd = $lastSession.started -}} {{- with $lastSession.finished -}}{{- $activityEnd = . -}}{{- end -}} {{- else -}} {{- with $entry.finished -}}{{- $activityEnd = . -}}{{- end -}} {{- end -}} {{- if or (eq $groupLast "") (gt $activityEnd $groupLast) -}} {{- $groupLast = $activityEnd -}} {{- end -}} {{- $start := $entry.started | time.Format $fmt -}} {{- $end := "" -}} {{- with $entry.finished -}}{{- $end = (. | time.Format $fmt) -}}{{- end -}} {{- if and (eq $end "") (not (in $openEndedStatuses $status)) -}} {{- $end = ($activityEnd | time.Format $fmt) -}} {{- end -}} {{- if or (eq $earliestStart "") (lt $start $earliestStart) -}} {{- $earliestStart = $start -}} {{- end -}} {{- $titleParts := slice ($entry.started | time.Format "Jan 2, 2006") -}} {{- if $entry.finished -}} {{- $titleParts = $titleParts | append ($entry.finished | time.Format "Jan 2, 2006") -}} {{- else -}} {{- $titleParts = $titleParts | append "ongoing" -}} {{- end -}} {{- with $entry.rating -}}{{- $titleParts = $titleParts | append (printf "%v/10" .) -}}{{- end -}} {{- $statusLabel := ($status | replace "_" " ") | title -}} {{- with $entry.subgame -}}{{- $statusLabel = printf "%s — %s" . $statusLabel -}}{{- end -}} {{- with $entry.platform -}}{{- $statusLabel = printf "%s (%s)" $statusLabel . -}}{{- end -}} {{- $title := printf "%s · %s" $statusLabel (delimit $titleParts " – ") -}} {{- with $entry.notes -}} {{- $title = printf "%s
%s" $title (. | markdownify | plainify | truncate 160) -}} {{- end -}} {{- $permalink := "" -}} {{- if $entry.written -}}{{- $permalink = $entry.permalink -}}{{- end -}} {{- if gt (len $entry.sessions) 0 -}} {{- $subgroupStack = merge $subgroupStack (dict $sub false) -}} {{/* No connector item emitted here: games-timeline.ts synthesizes connector bridges itself, splitting wherever the gap between sessions exceeds CONNECTOR_GAP_DAYS, so a years-long dormant stretch inside one long-running entry (Jackbox, Tabletop Simulator) doesn't get painted as one continuous dim bar. */}} {{- range $sIdx, $session := (sort $entry.sessions "started" "asc") -}} {{- $sStart := $session.started | time.Format $fmt -}} {{- $sEnd := "" -}} {{- with $session.finished -}}{{- $sEnd = (. | time.Format $fmt) -}}{{- end -}} {{- if and (eq $sEnd "") (not (in $openEndedStatuses $status)) -}} {{- $sEnd = $sStart -}} {{- end -}} {{- $items = $items | append (dict "id" (printf "%s--session-%d" $baseID $sIdx) "group" $slug "subgroup" $sub "type" "range" "className" (printf "pt-session pt--%s" $status) "content" "" "title" $title "status" ($status | replace "_" " " | title) "start" $sStart "end" $sEnd ) -}} {{- end -}} {{- else -}} {{- $items = $items | append (dict "id" $baseID "group" $slug "subgroup" $sub "type" "range" "className" (printf "pt pt--%s" $status) "content" "" "title" $title "status" ($status | replace "_" " " | title) "permalink" $permalink "start" $start "end" $end ) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $hasItems -}} {{/* A refreshed archive (achievements/playtime) can be more recent than any logged session — see game-last-played.html. */}} {{- with $game.Resources.Get "achievement-summary.yaml" -}} {{- with (. | transform.Unmarshal).last_played -}} {{- if gt . $groupLast -}}{{- $groupLast = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $groupCandidates = $groupCandidates | append (dict "id" $slug "content" (printf "%s" $game.RelPermalink $game.Title) "subgroupStack" $subgroupStack "sortKey" $groupLast ) -}} {{- end -}} {{- end -}} {{/* Row order matches the list below: most-recently-played game at the top. */}} {{- $groups := slice -}} {{- range sort $groupCandidates "sortKey" "desc" -}} {{- $groups = $groups | append (dict "id" .id "content" .content "subgroupStack" .subgroupStack) -}} {{- end
