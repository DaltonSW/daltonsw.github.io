{{/* Context: a game's term Page. Returns the most recent known activity date as a "2006-01-02" string: the max of each entry's last known touch-point and achievement-summary.yaml's last_played (Steam's real timestamp, or RA's last-unlock date as its only proxy — see writeAchievementSummary in tools/gamelog). "" when neither exists — a game's front matter carries no dates. */}} {{- $entries := partial "playthrough-entries.html" . -}} {{- $latest := "" -}} {{- range $entries -}} {{- $candidate := (partial "entry-last-activity.html" .).end -}} {{- if and $candidate (or (eq $latest "") (gt $candidate $latest)) -}} {{- $latest = $candidate -}} {{- end -}} {{- end -}} {{- with .Resources.Get "achievement-summary.yaml" -}} {{- with (. | transform.Unmarshal).last_played -}} {{- if or (eq $latest "") (gt . $latest) -}}{{- $latest = . -}}{{- end -}} {{- end -}} {{- end -}} {{- return $latest -}}
