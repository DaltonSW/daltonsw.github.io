{{/* Context: a game's term Page. Returns the most recent known activity date as a "2006-01-02" string: the max of each entry's last known touch-point and achievement-summary.yaml's last_played (Steam's real timestamp, or RA's last-unlock date as its only proxy — see writeAchievementSummary in tools/gamelog). Falls back to front-matter only when nothing else exists. */}} {{- $entries := partial "playthrough-entries.html" . -}} {{- $latest := "" -}} {{- range $entries -}} {{- $candidate := "" -}} {{- if gt (len .sessions) 0 -}} {{- $sorted := sort .sessions "started" "asc" -}} {{- $last := index $sorted (sub (len $sorted) 1) -}} {{- $candidate = $last.started -}} {{- with $last.finished -}}{{- $candidate = . -}}{{- end -}} {{- else -}} {{- $candidate = .started -}} {{- with .finished -}}{{- $candidate = . -}}{{- end -}} {{- end -}} {{- if and $candidate (or (eq $latest "") (gt $candidate $latest)) -}} {{- $latest = $candidate -}} {{- end -}} {{- end -}} {{- with .Resources.Get "achievement-summary.yaml" -}} {{- with (. | transform.Unmarshal).last_played -}} {{- if or (eq $latest "") (gt . $latest) -}}{{- $latest = . -}}{{- end -}} {{- end -}} {{- end -}} {{- if eq $latest "" -}} {{- with .Params.finished -}}{{- $latest = (. | time.Format "2006-01-02") -}}{{- end -}} {{- if eq $latest "" -}} {{- with .Params.started -}}{{- $latest = (. | time.Format "2006-01-02") -}}{{- end -}} {{- end -}} {{- end -}} {{- return $latest -}}
