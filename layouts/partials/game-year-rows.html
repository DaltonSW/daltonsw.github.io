{{/* Context: a game's term Page. Returns one row dict per calendar year the game's playthroughs touched, newest first (year/count/start/end/status/ platform). A playthrough lands in every year it touched, and rows merge per game per year, not per playthrough. */}} {{- $game := . -}} {{- $entries := partial "playthrough-entries.html" $game -}} {{- if eq (len $entries) 0 -}} {{/* normalize possible time.Time (from hugo.toml [frontmatter] wiring) to a string */}} {{- with $game.Params.started -}} {{- $finished := "" -}} {{- with $game.Params.finished -}}{{- $finished = (. | time.Format "2006-01-02") -}}{{- end -}} {{- $entries = slice (dict "started" (. | time.Format "2006-01-02") "finished" $finished "status" $game.Params.status "platform" $game.Params.platform "sessions" slice ) -}} {{- end -}} {{- end -}} {{- $rows := dict -}} {{- range $entries -}} {{- if .started -}} {{/* normalize to "2006-01-02" strings before any comparison or substr */}} {{- $started := .started | time.Format "2006-01-02" -}} {{- $finished := "" -}} {{- with .finished -}}{{- $finished = (. | time.Format "2006-01-02") -}}{{- end -}} {{/* same last-activity rule as game-last-played.html and the timeline */}} {{- $end := $started -}} {{- if gt (len .sessions) 0 -}} {{- $sorted := sort .sessions "started" "asc" -}} {{- $last := index $sorted (sub (len $sorted) 1) -}} {{- $end = $last.started -}} {{- with $last.finished -}}{{- $end = . -}}{{- end -}} {{- else -}} {{- with $finished -}}{{- $end = . -}}{{- end -}} {{- end -}} {{/* span this entry contributes to each year, clipped to that year */}} {{- $spans := dict -}} {{- $ranges := slice (dict "start" $started "end" $end) -}} {{- if gt (len .sessions) 0 -}} {{- $ranges = slice -}} {{- range .sessions -}} {{- $ranges = $ranges | append (dict "start" .started "end" (.finished | default .started)) -}} {{- end -}} {{- end -}} {{- range $ranges -}} {{- $rs := .start -}} {{- $re := .end -}} {{- range seq (int (substr $rs 0 4)) (int (substr $re 0 4)) -}} {{- $year := printf "%d" . -}} {{- $cs := $rs -}} {{- $ce := $re -}} {{- if lt $cs (printf "%s-01-01" $year) -}}{{- $cs = printf "%s-01-01" $year -}}{{- end -}} {{- if gt $ce (printf "%s-12-31" $year) -}}{{- $ce = printf "%s-12-31" $year -}}{{- end -}} {{- $span := index $spans $year -}} {{- if not $span -}} {{- $span = dict "start" $cs "end" $ce -}} {{- end -}} {{- if lt $cs $span.start -}}{{- $span = merge $span (dict "start" $cs) -}}{{- end -}} {{- if gt $ce $span.end -}}{{- $span = merge $span (dict "end" $ce) -}}{{- end -}} {{- $spans = merge $spans (dict $year $span) -}} {{- end -}} {{- end -}} {{- $entry := . -}} {{- range $year, $span := $spans -}} {{- $row := index $rows $year -}} {{- if not $row -}} {{- $row = dict "year" $year "count" 0 "start" $span.start "end" $span.end "status" $entry.status "platform" $entry.platform "latest" $span.end -}} {{- end -}} {{- $row = merge $row (dict "count" (add $row.count 1)) -}} {{- if lt $span.start $row.start -}} {{- $row = merge $row (dict "start" $span.start) -}} {{- end -}} {{- if gt $span.end $row.end -}} {{- $row = merge $row (dict "end" $span.end) -}} {{- end -}} {{/* status/platform come from whichever contributor ran latest */}} {{- if ge $span.end $row.latest -}} {{- $row = merge $row (dict "latest" $span.end "status" $entry.status "platform" $entry.platform) -}} {{- end -}} {{- $rows = merge $rows (dict $year $row) -}} {{- end -}} {{- end -}} {{- end -}} {{/* endless/multiplayer/software are game-level statuses; let them override the entry's own "playing" status (see tools/gamelog KNOWN-ISSUES) */}} {{- $gameStatus := $game.Params.status -}} {{- $override := in (slice "endless" "multiplayer" "software") $gameStatus -}} {{- $out := slice -}} {{- range $year, $row := $rows -}} {{- $status := $row.status | default $gameStatus -}} {{- if $override -}}{{- $status = $gameStatus -}}{{- end -}} {{- $out = $out | append (merge $row (dict "status" $status)) -}} {{- end -}} {{- return (sort $out "year" "desc") -}}
