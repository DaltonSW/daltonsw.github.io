{{/* Context: a game's term Page. Returns the earliest known "started" date as a "2006-01-02" string, falling back to front-matter Params.started (same fallback games-timeline.html uses) only when nothing's logged yet. */}} {{- $entries := partial "playthrough-entries.html" . -}} {{- $earliest := "" -}} {{- range $entries -}} {{- if .started -}} {{- if or (eq $earliest "") (lt .started $earliest) -}} {{- $earliest = .started -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $earliest "" -}} {{- with .Params.started -}}{{- $earliest = (. | time.Format "2006-01-02") -}}{{- end -}} {{- end -}} {{- return $earliest -}}
