{{/* Context: a game's term Page. Returns the day the game was first beaten as a "2006-01-02" string: the earliest finish date among entries with status "finished" (dropped/unfinished runs ended, but nothing was finished). "" when there's none. Mirrors firstFinished in tools/gamelog/internal/model. */}} {{- $first := "" -}} {{- range partial "playthrough-entries.html" . -}} {{- if and (eq .status "finished") .finished -}} {{- $d := .finished | time.Format "2006-01-02" -}} {{- if or (eq $first "") (lt $d $first) -}}{{- $first = $d -}}{{- end -}} {{- end -}} {{- end -}} {{- return $first -}}
