Returns entries merging the game's logged playthroughs (playthroughs.yaml, rewritten whole by the tool) with its nested written playthrough pages, normalized into a common shape. */}} {{- $entries := slice -}} {{- $logged := slice -}} {{- with .Resources.Get "playthroughs.yaml" -}} {{- with (. | transform.Unmarshal).playthroughs -}} {{- $logged = . -}} {{- end -}} {{- end -}} {{- range $logged }} {{- $started := .started }} {{- $finished := .finished }} {{- with .sessions }} {{- $sorted := sort .
